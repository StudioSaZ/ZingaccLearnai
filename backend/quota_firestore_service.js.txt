  Future<void> _addCoinsToUserAccount(String packageName, int packageCoins) async {
    try {
      final user = FirebaseAuth.instance.currentUser;
      if (user == null) return;

      final userRef = FirebaseFirestore.instance.collection('users').doc(user.uid);

      await userRef.set({
        'quota': {
          'paidBalance': FieldValue.increment(packageCoins),
          'lastPaymentDate': FieldValue.serverTimestamp(),
          'activePackage': packageName,
        }
      }, SetOptions(merge: true));

      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text('ගෙවීම සාර්ථකයි! AI ප්‍රශ්න $packageCoins ක් ඔබේ ගිණුමට එකතු විය. 🎉'),
          backgroundColor: Colors.green,
          duration: const Duration(seconds: 4),
        ),
      );
    } catch (e) {
      debugPrint("Error updating coins: $e");
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('ගෙවීම සාර්ථක වුවද ප්‍රශ්න යාවත්කාලීන කිරීමේදී දෝෂයක් විය. කරුණාකර අප හා සම්බන්ධ වන්න.'),
          backgroundColor: Colors.orange,
        ),
      );
    }
  }
